> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Index `.pptx`, `.odt`, `.ods`, `.epub`, `.doc`, `.rtf`, `.srt`, `.vtt` (local and, except subtitles, cloud), close the cloud `.xls`/generic-MIME `.csv`/`.html` gap, and give RTF-only `.msg` bodies their text.

**Architecture:** Each format is one new `ConvertibleKind` with a pure parser module under `src/main/core/converter/formats/`, dispatched from `parseOther` in `parsers.ts` (runs in the converter child). **Tech Stack:** TypeScript, jest, jszip 3.10 + @xmldom/xmldom 0.8 (already installed via mammoth; declared directly here), SheetJS 0.20.3 (ODS), `word-extractor` 1.0.4 (new, MIT), `@kenjiuno/decompressrtf` (already installed via msgreader), turndown.

**Spec:** `docs/superpowers/specs/2026-10-13-file-formats-batch-design.md` — section A, Rollout step 1, Re-enumeration row "Phase 1".

## Global Constraints

- `parsers.ts` and everything under `converter/formats/` import nothing from electron, the store, the engine or the log sink (existing test enforces this for `parsers.ts`; Task 1 extends it to `formats/`).
- Output stays capped by `MAX_MARKDOWN_CHARS` (2 MiB) — the existing `capMarkdown` call sites apply; parsers may stop early once they exceed it.
- Zip safety (OOXML/ODF/EPUB): read only named parts; refuse a part whose uncompressed size exceeds 20 MiB; stop after 50 MiB inflated in total; never inflate media.
- First markdown line is `# <title>` (document title from properties, else none — the row's title already shows the filename).
- A parser throw records `conversion.status = 'failed'` exactly like DOCX today (no new failure path).
- Size caps and the `docCap` ladder are unchanged.
- `FILE_POLICY_VERSION = 3`; `ADMITTED_SINCE[3] = pptx, odt, ods, epub, doc, rtf, srt, vtt` (amended after the staged review: csv/xls/html were already indexed locally).
- Run builds/tests sequentially, never in parallel. Jest needs `npm run build:main && npm run build:renderer` to have run once in the worktree.
- Commits: no `Co-Authored-By` line; use `git commit -F <file>` when the message has backticks.

## Review Focus

1. A `.pptx` whose `presentation.xml` lists a slide whose part is missing, or a hidden/odd rels layout → the remaining slides still render; never a throw for one missing slide. (Task 2 test.)
2. Bytes that are not a zip at all but carry `.pptx/.odt/.epub` → a throw (→ `failed`), never a hang or a partial garbage render. (Task 1 test.)
3. RTF in a non-Latin codepage (`\ansicpg1251`, `\'xx`) and `\uN?` escapes (German/Russian users) → correct Unicode text. (Task 5 test.)
4. Subtitles with a UTF-8 BOM, CRLF line endings, VTT `NOTE`/`STYLE` blocks and inline tags → clean text with minute markers. (Task 6 test.)
5. EPUB whose OPF sits in a subdirectory (`OEBPS/content.opf`) with spine hrefs relative to it, plus URL-encoded hrefs → chapters found in spine order. (Task 3 test.)

```ts
/** Bounded reads of named parts from an OOXML/ODF/EPUB zip. Only the parts a
 *  parser asks for are inflated; media never are. Pure (converter child). */
import JSZip from 'jszip';

export const MAX_ZIP_PART_BYTES = 20 * 1024 * 1024;
export const MAX_ZIP_TOTAL_BYTES = 50 * 1024 * 1024;

export interface ZipParts {
  has(name: string): boolean;
  text(name: string): Promise<string | null>;
}

/** JSZip keeps the central-directory size on an internal field. */
const declaredSize = (f: JSZip.JSZipObject): number | null => {
  const n = (f as unknown as { _data?: { uncompressedSize?: unknown } })._data
    ?.uncompressedSize;
  return typeof n === 'number' && Number.isFinite(n) ? n : null;
};

export async function openZipParts(bytes: Uint8Array): Promise<ZipParts> {
  const zip = await JSZip.loadAsync(bytes);
  let inflated = 0;
  const charge = (name: string, n: number) => {
    if (n > MAX_ZIP_PART_BYTES) throw new Error(`zip part too large: ${name}`);
    inflated += n;
    if (inflated > MAX_ZIP_TOTAL_BYTES)
      throw new Error('zip inflate budget exceeded');
  };
  return {
    has: (name) => zip.file(name) != null,
    async text(name) {
      const f = zip.file(name);
      if (!f) return null;
      const declared = declaredSize(f);
      if (declared != null) charge(name, declared);
      const s = await f.async('string');
      if (declared == null) charge(name, s.length);
      return s;
    },
  };
}
```

```ts
/** Namespace-agnostic XML helpers over @xmldom/xmldom. Pure. */
import { DOMParser } from '@xmldom/xmldom';

export function parseXml(src: string): Document {
  return new DOMParser({
    onError: () => {},
  } as never).parseFromString(src, 'text/xml') as unknown as Document;
}

export function byLocalName(root: Node, name: string): Element[] {
  const out: Element[] = [];
  const walk = (n: Node) => {
    for (let c = n.firstChild; c; c = c.nextSibling) {
      if (c.nodeType === 1) {
        if ((c as Element).localName === name) out.push(c as Element);
        walk(c);
      }
    }
  };
  walk(root);
  return out;
}

export function childByLocalName(el: Element, name: string): Element | null {
  for (let c = el.firstChild; c; c = c.nextSibling)
    if (c.nodeType === 1 && (c as Element).localName === name)
      return c as Element;
  return null;
}

/** Text of a node, with ODF (`s`, `tab`, `line-break`) and OOXML (`br`,
 *  `tab`) spacing elements rendered as whitespace. */
export function textContent(node: Node): string {
  let out = '';
  for (let c = node.firstChild; c; c = c.nextSibling) {
    if (c.nodeType === 3 || c.nodeType === 4) out += c.nodeValue ?? '';
    else if (c.nodeType === 1) {
      const el = c as Element;
      switch (el.localName) {
        case 's': {
          const n = Number(
            el.getAttribute('text:c') ?? el.getAttribute('c') ?? '1',
          );
          out += ' '.repeat(Number.isFinite(n) && n > 0 ? n : 1);
          break;
        }
        case 'tab':
          out += '\t';
          break;
        case 'line-break':
        case 'br':
          out += '\n';
          break;
        default:
          out += textContent(el);
      }
    }
  }
  return out;
}
```

```ts
const XML = '<?xml version="1.0" encoding="UTF-8"?>';
const P = 'http://schemas.openxmlformats.org/presentationml/2006/main';
const A = 'http://schemas.openxmlformats.org/drawingml/2006/main';
const R = 'http://schemas.openxmlformats.org/officeDocument/2006/relationships';
const REL = 'http://schemas.openxmlformats.org/package/2006/relationships';

export interface SlideSpec {
  title?: string;
  body?: string[];
  notes?: string[];
}

const sp = (paras: string[], ph?: string) =>
  `<p:sp><p:nvSpPr><p:cNvPr id="1" name="s"/><p:cNvSpPr/><p:nvPr>${
    ph ? `<p:ph type="${ph}"/>` : ''
  }</p:nvPr></p:nvSpPr><p:txBody>${paras
    .map((t) => `<a:p><a:r><a:t>${t}</a:t></a:r></a:p>`)

export async function pptxBytes(
  slides: SlideSpec[],
  opts: { title?: string; author?: string; dropSlide?: number } = {},

  zip.file(

      )
      .join('')}</Relationships>`,

    if (opts.dropSlide === n) return;

const ODF_NS =
  'xmlns:office="urn:oasis:names:tc:opendocument:xmlns:office:1.0" xmlns:text="urn:oasis:names:tc:opendocument:xmlns:text:1.0" xmlns:table="urn:oasis:names:tc:opendocument:xmlns:table:1.0"';

export async function odtBytes(
  body: string,
  title?: string,
): Promise<Uint8Array> {
  const zip = new JSZip();
  zip.file('mimetype', 'application/vnd.oasis.opendocument.text');
  zip.file(
    'content.xml',
    `${XML}<office:document-content ${ODF_NS}><office:body><office:text>${body}</office:text></office:body></office:document-content>`,

  if (title)
    zip.file(
      'meta.xml',

export interface EpubSpec {
  opfDir: string; // e.g. 'OEBPS' or ''
  title?: string;
  author?: string;
  chapters: Array<{ href: string; html: string }>; // spine order
}

export async function epubBytes(spec: EpubSpec): Promise<Uint8Array> {
  const zip = new JSZip();
  const opfPath = spec.opfDir ? `${spec.opfDir}/content.opf` : 'content.opf';
  zip.file('mimetype', 'application/epub+zip');
  zip.file(
    'META-INF/container.xml',
    `${XML}<container xmlns="urn:oasis:names:tc:opendocument:xmlns:container" version="1.0"><rootfiles><rootfile full-path="${opfPath}" media-type="application/oebps-package+xml"/></rootfiles></container>`,
  );
  zip.file(
    opfPath,
    `${XML}<package xmlns="http://www.idpf.org/2007/opf" version="3.0"><metadata xmlns:dc="http://purl.org/dc/elements/1.1/">${
      spec.title ? `<dc:title>${spec.title}</dc:title>` : ''
    }${spec.author ? `<dc:creator>${spec.author}</dc:creator>` : ''}</metadata><manifest>${spec.chapters
      .map(
        (c, i) =>
          `<item id="c${i}" href="${c.href}" media-type="application/xhtml+xml"/>`,
      )
      .join('')}</manifest><spine>${spec.chapters
      .map((_, i) => `<itemref idref="c${i}"/>`)
      .join('')}</spine></package>`,
  );
  for (const c of spec.chapters) {
    const p = decodeURIComponent(c.href);
    zip.file(spec.opfDir ? `${spec.opfDir}/${p}` : p, c.html);
  }
  return zip.generateAsync({ type: 'uint8array' });
}

export async function odsBytes(rows: string[][]): Promise<Uint8Array> {
  const XLSX = await import('xlsx');
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(rows), 'Budget');
  return new Uint8Array(XLSX.write(wb, { type: 'buffer', bookType: 'ods' }));
}
```

```ts
/** PowerPoint OOXML → markdown: slides in presentation order, each with its
 *  title placeholder as a heading, the remaining text, then speaker notes.
 *  Only slide, notes, rels and core-property parts are inflated. Pure. */
import { openZipParts, type ZipParts } from './zip-parts';
import { byLocalName, childByLocalName, parseXml, textContent } from './xml';

const TITLE_TYPES = new Set(['title', 'ctrTitle']);
const NOTES_SKIP = new Set(['sldNum', 'sldImg', 'hdr', 'ftr', 'dt']);

const phType = (sp: Element): string | null => {
  const nv = childByLocalName(sp, 'nvSpPr');
  const nvPr = nv && childByLocalName(nv, 'nvPr');
  const ph = nvPr && childByLocalName(nvPr, 'ph');
  return ph ? (ph.getAttribute('type') ?? 'body') : null;
};

const paragraphs = (el: Element): string[] =>
  byLocalName(el, 'p')
    .filter((p) => p.namespaceURI?.includes('drawingml'))
    .map((p) => textContent(p).trim())
    .filter(Boolean);

/** Relationship Id → resolved part path, relative to `base` (a part path). */
async function rels(
  parts: ZipParts,
  base: string,
): Promise<Map<string, { type: string; target: string }>> {
  const dir = base.slice(0, base.lastIndexOf('/'));
  const name = base.slice(base.lastIndexOf('/') + 1);
  const xml = await parts.text(`${dir}/_rels/${name}.rels`);
  const out = new Map<string, { type: string; target: string }>();
  if (!xml) return out;
  for (const r of byLocalName(parseXml(xml), 'Relationship')) {
    const target = r.getAttribute('Target') ?? '';
    const segs = `${dir}/${target}`.split('/');
    const resolved: string[] = [];
    for (const s of segs) {
      if (s === '..') resolved.pop();
      else if (s && s !== '.') resolved.push(s);
    }
    out.set(r.getAttribute('Id') ?? '', {
      type: r.getAttribute('Type') ?? '',
      target: target.startsWith('/') ? target.slice(1) : resolved.join('/'),
    });
  }
  return out;
}

export async function pptxToMarkdown(bytes: Uint8Array): Promise<string> {
  const parts = await openZipParts(bytes);
  const pres = await parts.text('ppt/presentation.xml');
  if (!pres) throw new Error('pptx: no ppt/presentation.xml');
  const presRels = await rels(parts, 'ppt/presentation.xml');
  const out: string[] = [];

  const core = await parts.text('docProps/core.xml');
  if (core) {
    const doc = parseXml(core);
    const title = byLocalName(doc, 'title')[0];
    const author = byLocalName(doc, 'creator')[0];
    if (title && textContent(title).trim())
      out.push(`# ${textContent(title).trim()}`);
    if (author && textContent(author).trim())
      out.push(`**Author:** ${textContent(author).trim()}`);
  }

  const ids = byLocalName(parseXml(pres), 'sldId');
  for (let i = 0; i < ids.length; i++) {
    let rid = '';
    for (let a = 0; a < ids[i].attributes.length; a++) {
      const at = ids[i].attributes.item(a)!;
      if (at.localName === 'id' && at.prefix === 'r') rid = at.value;
    }
    const part = presRels.get(rid)?.target;
    const xml = part ? await parts.text(part) : null;
    if (!part || !xml) continue;
    const shapes = byLocalName(parseXml(xml), 'sp');
    const titleSp = shapes.find((s) => TITLE_TYPES.has(phType(s) ?? ''));
    const title = titleSp ? paragraphs(titleSp).join(' ') : '';
    const body = shapes
      .filter((s) => s !== titleSp)
      .flatMap(paragraphs);
    // Tables live in graphicFrames, not sp: take their cell paragraphs too.
    for (const tc of byLocalName(parseXml(xml), 'tc'))
      body.push(...paragraphs(tc));
    out.push(title ? `## Slide ${i + 1} — ${title}` : `## Slide ${i + 1}`);
    if (body.length) out.push(body.join('\n'));

    const notesRel = [...(await rels(parts, part)).values()].find((r) =>
      r.type.endsWith('/notesSlide'),
    );
    const notesXml = notesRel ? await parts.text(notesRel.target) : null;
    if (notesXml) {
      const notes = byLocalName(parseXml(notesXml), 'sp')
        .filter((s) => !NOTES_SKIP.has(phType(s) ?? ''))
        .flatMap(paragraphs);
      if (notes.length) out.push('### Notes', notes.join('\n'));
    }
  }
  return out.join('\n\n');
}
```

```ts
/** OpenDocument text → markdown from content.xml (+ meta.xml title). Pure. */
import { openZipParts } from './zip-parts';
import { byLocalName, parseXml, textContent } from './xml';

function block(el: Element): string | null {
  switch (el.localName) {
    case 'h': {
      let level = 1;
      for (let i = 0; i < el.attributes.length; i++) {
        const a = el.attributes.item(i)!;
        if (a.localName === 'outline-level') level = Number(a.value) || 1;
      }
      const t = textContent(el).trim();
      return t ? `${'#'.repeat(Math.min(level, 6))} ${t}` : null;
    }
    case 'p': {
      const t = textContent(el).trim();
      return t || null;
    }
    case 'list':
      return (
        byLocalName(el, 'list-item')
          .map((li) => {
            const t = byLocalName(li, 'p')
              .map((p) => textContent(p).trim())
              .filter(Boolean)
              .join(' ');
            return t ? `- ${t}` : '';
          })
          .filter(Boolean)
          .join('\n') || null
      );
    case 'table':
      return (
        byLocalName(el, 'table-row')
          .map((row) =>
            byLocalName(row, 'table-cell')
              .map((c) => textContent(c).trim())
              .join(' | '),
          )
          .filter((r) => r.replace(/[|\s]/g, ''))
          .join('\n') || null
      );
    case 'section':
      return blocks(el).join('\n\n') || null;
    default:
      return null;
  }
}

function blocks(parent: Element): string[] {
  const out: string[] = [];
  for (let c = parent.firstChild; c; c = c.nextSibling)
    if (c.nodeType === 1) {
      const b = block(c as Element);
      if (b) out.push(b);
    }
  return out;
}

export async function odtToMarkdown(bytes: Uint8Array): Promise<string> {
  const parts = await openZipParts(bytes);
  const content = await parts.text('content.xml');
  if (!content) throw new Error('odt: no content.xml');
  const out: string[] = [];
  const meta = await parts.text('meta.xml');
  const title = meta ? byLocalName(parseXml(meta), 'title')[0] : undefined;
  if (title && textContent(title).trim())
    out.push(`# ${textContent(title).trim()}`);
  const text = byLocalName(parseXml(content), 'text').find(
    (e) => e.namespaceURI?.endsWith(':office:1.0'),
  );
  if (text) out.push(...blocks(text));
  return out.join('\n\n');
}
```

```ts
/** EPUB → markdown: OPF metadata, then spine XHTML in order via the shared
 *  HTML→markdown converter. Stops once the markdown cap is reached. Pure. */
import { MAX_MARKDOWN_CHARS } from '../parsers';
import { openZipParts } from './zip-parts';
import { byLocalName, parseXml, textContent } from './xml';

const join = (dir: string, href: string): string => {
  const out: string[] = [];
  for (const s of `${dir}/${decodeURIComponent(href.split('#')[0])}`.split('/')) {
    if (s === '..') out.pop();
    else if (s && s !== '.') out.push(s);
  }
  return out.join('/');
};

export async function epubToMarkdown(
  bytes: Uint8Array,
  html: (s: string) => Promise<string>,
): Promise<string> {
  const parts = await openZipParts(bytes);
  const container = await parts.text('META-INF/container.xml');
  if (!container) throw new Error('epub: no META-INF/container.xml');
  const rootfile = byLocalName(parseXml(container), 'rootfile')[0];
  const opfPath = rootfile?.getAttribute('full-path') ?? '';
  const opf = opfPath ? await parts.text(opfPath) : null;
  if (!opf) throw new Error('epub: no package document');
  const doc = parseXml(opf);
  const dir = opfPath.includes('/')
    ? opfPath.slice(0, opfPath.lastIndexOf('/'))
    : '';
  const out: string[] = [];
  const meta = (n: string) => {
    const e = byLocalName(doc, n)[0];
    return e ? textContent(e).trim() : '';
  };
  if (meta('title')) out.push(`# ${meta('title')}`);
  if (meta('creator')) out.push(`**Author:** ${meta('creator')}`);
  const hrefs = new Map(
    byLocalName(doc, 'item').map((i) => [
      i.getAttribute('id') ?? '',
      i.getAttribute('href') ?? '',
    ]),
  );
  let size = out.join('\n\n').length;
  for (const ref of byLocalName(doc, 'itemref')) {
    const href = hrefs.get(ref.getAttribute('idref') ?? '');
    if (!href) continue;
    const xhtml = await parts.text(join(dir, href));
    if (!xhtml) continue;
    const md = (await html(xhtml)).trim();
    if (!md) continue;
    out.push(md);
    size += md.length;
    if (size > MAX_MARKDOWN_CHARS) break;
  }
  return out.join('\n\n');
}
```

```ts
/** Word 97–2003 `.doc` → text via word-extractor (pure JS CFB reader).
 *  Body, then headers/footers, footnotes, endnotes and comments when present.
 *  Pure. */
export async function docToMarkdown(bytes: Uint8Array): Promise<string> {
  // CJS default export; typed lazy require like msgreader in parsers.ts.
  // eslint-disable-next-line @typescript-eslint/no-require-imports
  const WordExtractor = require('word-extractor') as new () => {
    extract(input: Buffer): Promise<{
      getBody(): string;
      getHeaders(o?: { includeFooters?: boolean }): string;
      getFootnotes(): string;
      getEndnotes(): string;
      getAnnotations(): string;
    }>;
  };
  const doc = await new WordExtractor().extract(Buffer.from(bytes));
  const sections: Array<[string, string]> = [
    ['', doc.getBody()],
    ['Headers and footers', doc.getHeaders({ includeFooters: true })],
    ['Footnotes', doc.getFootnotes()],
    ['Endnotes', doc.getEndnotes()],
    ['Comments', doc.getAnnotations()],
  ];
  return sections
    .map(([h, t]) => [h, t.replace(/\r\n?/g, '\n').trim()] as const)
    .filter(([, t]) => t)
    .map(([h, t]) => (h ? `## ${h}\n\n${t}` : t))
    .join('\n\n');
}
```

```ts
/** MS-OXRTFCP uncompressed container ("MELA"): header + raw RTF bytes. */
export function compressRTF(rtf: string): Uint8Array {
  const raw = Buffer.from(rtf, 'latin1');
  const head = Buffer.alloc(16);
  head.writeUInt32LE(raw.length + 12, 0); // COMPSIZE: size after this field
```

```ts
/** Minimal RTF → plain text. Handles groups, control words, \'hh in the
 *  document codepage, \uN with \ucN fallback skipping, \bin, escaped
 *  symbols; skips non-text destinations (fonts, colors, styles, info,
 *  pictures, \* destinations, field instructions). Pure. */

const SKIP_DEST = new Set([
  'fonttbl', 'colortbl', 'stylesheet', 'info', 'pict', 'object',
  'themedata', 'colorschememapping', 'datastore', 'latentstyles',
  'listtable', 'listoverridetable', 'rsidtbl', 'generator', 'xmlnstbl',
  'fldinst', 'header', 'footer', 'headerl', 'headerr', 'footerl',
  'footerr', 'filetbl', 'revtbl', 'pgdsctbl', 'expandedcolortbl',
]);

interface Group {
  skip: boolean;
  uc: number;
}

const decoderFor = (cp: number): TextDecoder => {
  try {
    return new TextDecoder(`windows-${cp}`);
  } catch {
    return new TextDecoder('windows-1252');
  }
};

export function rtfToText(src: string): string {
  let out = '';
  let pending: number[] = []; // \'hh bytes awaiting decode
  let decoder = decoderFor(1252);
  const stack: Group[] = [];
  let g: Group = { skip: false, uc: 1 };
  let skipChars = 0; // \uN fallback chars to drop
  let i = 0;

  const flush = () => {
    if (pending.length) {
      if (!g.skip) out += decoder.decode(new Uint8Array(pending));
      pending = [];
    }
  };
  const emit = (s: string) => {
    flush();
    if (g.skip) return;
    if (skipChars > 0) {
      const drop = Math.min(skipChars, s.length);
      skipChars -= drop;
      s = s.slice(drop);
    }
    out += s;
  };

  while (i < src.length) {
    const ch = src[i];
    if (ch === '{') {
      flush();
      stack.push(g);
      g = { ...g };
      i++;
      // `{\*\dest` — an ignorable destination.
      if (src.startsWith('\\*', i)) g.skip = true;
      continue;
    }
    if (ch === '}') {
      flush();
      g = stack.pop() ?? { skip: false, uc: 1 };
      i++;
      continue;
    }
    if (ch === '\r' || ch === '\n') {
      i++;
      continue;
    }
    if (ch !== '\\') {
      emit(ch);
      i++;
      continue;
    }
    // Control symbol or word.
    const next = src[i + 1];
    if (next === undefined) break;
    if (next === '\\' || next === '{' || next === '}') {
      emit(next);
      i += 2;
      continue;
    }
    if (next === "'") {
      const hex = src.slice(i + 2, i + 4);
      i += 4;
      if (skipChars > 0) {
        skipChars--;
        continue;
      }
      if (!g.skip) pending.push(parseInt(hex, 16));
      continue;
    }
    if (next === '~') {
      emit(' ');
      i += 2;
      continue;
    }
    if (next === '-' || next === '*') {
      i += 2;
      continue;
    }
    if (next === '_') {
      emit('-');
      i += 2;
      continue;
    }
    const m = /^([a-zA-Z]+)(-?\d+)? ?/.exec(src.slice(i + 1, i + 40));
    if (!m) {
      i += 2;
      continue;
    }
    i += 1 + m[0].length;
    const word = m[1];
    const arg = m[2] === undefined ? null : Number(m[2]);
    if (SKIP_DEST.has(word)) {
      g.skip = true;
      continue;
    }
    switch (word) {
      case 'par':
      case 'line':
      case 'sect':
      case 'page':
        emit('\n');
        break;
      case 'tab':
        emit('\t');
        break;
      case 'ansicpg':
        if (arg) decoder = decoderFor(arg);
        break;
      case 'uc':
        g.uc = arg ?? 1;
        break;
      case 'u': {
        const code = arg == null ? 0 : arg < 0 ? arg + 65536 : arg;
        emit(String.fromCharCode(code));
        skipChars = g.uc;
        break;
      }
      case 'bin':
        i += arg ?? 0;
        break;
      default:
        break;
    }
  }
  flush();
  return out
    .replace(/[ \t]+\n/g, '\n')
    .replace(/\n{3,}/g, '\n\n')
    .trim();
}

export function rtfBytesToMarkdown(bytes: Uint8Array): string {
  return rtfToText(Buffer.from(bytes).toString('latin1'));
}
```

```ts
/** SRT/VTT → a readable transcript: one paragraph per minute, prefixed
 *  `[mm:ss]` (minutes may exceed 59), cue numbers/timings/tags dropped,
 *  consecutive duplicate lines collapsed. Pure. */
const TIMING =
  /^\s*((?:\d+:)?\d{1,2}:\d{2}[.,]\d{1,3})\s*-->\s*(?:\d+:)?\d{1,2}:\d{2}[.,]\d{1,3}/;

const seconds = (t: string): number => {
  const parts = t.replace(',', '.').split(':').map(Number);
  return parts.reduce((acc, p) => acc * 60 + p, 0);
};

export function subtitlesToMarkdown(text: string): string {
  const blocks = text
    .replace(/^﻿/, '')
    .replace(/\r\n?/g, '\n')
    .split(/\n{2,}/);
  const minutes = new Map<number, string[]>();
  let last = '';
  for (const block of blocks) {
    const lines = block.split('\n');
    const at = lines.findIndex((l) => TIMING.test(l));
    if (at < 0) continue; // WEBVTT header, NOTE, STYLE, REGION
    const minute = Math.floor(seconds(TIMING.exec(lines[at])![1]) / 60);
    const cue = lines
      .slice(at + 1)
      .map((l) => l.replace(/<[^>]+>/g, '').trim())
      .filter(Boolean)
      .join(' ');
    if (!cue || cue === last) continue;
    last = cue;
    const list = minutes.get(minute) ?? [];
    list.push(cue);
    minutes.set(minute, list);
  }
  return [...minutes.entries()]
    .sort(([a], [b]) => a - b)
    .map(([m, cues]) => `[${String(m).padStart(2, '0')}:00] ${cues.join(' ')}`)
    .join('\n\n');
}
```
